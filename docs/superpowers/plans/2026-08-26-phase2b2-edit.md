# 薬局業務日誌 フェーズ2b-2：日誌編集 実装計画

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 既存日の日誌詳細画面から、その場で内容を修正して保存できるようにする。

**Architecture:** 既存の `src/app/views/detail.js`(閲覧)と対になる `src/app/views/edit.js`(編集)を新設する。`main.js` が詳細画面の「編集」ボタンで `EditView` に切り替え、「保存」で `DailyLog.normalize`→`DailyLog.validate`→`Storage.saveLog` の順に処理し、成功したら `DetailView` に戻す。

**Tech Stack:** Firebase JS SDK v10.8.0、素の JavaScript

## Global Constraints

- **依存パッケージを追加しない。** フェーズ1・2a・2b-1と同じ方針
- **`src/model/` と `src/parse/` の既存モジュールは書き換えない**
- **表示は既存の閲覧画面(detail.js)の見た目・カード構成・配色を踏襲する**
- 画面はスマートフォンでも読めること。横スクロールはテーブル単位に閉じ込め、ページ全体を横スクロールさせない
- **新規日の作成は対象外。** 既存日(Firestoreに既にドキュメントがある日)の編集のみ
- **マスタ編集は対象外。** `premixes`・`doctorChanges` は既存行の数値のみ編集可能とし、行の追加・削除は不可
- **`submissions`・`checklist` は編集対象外。** 現在の詳細画面に表示セクションが無く、過去データ移行でも値が入っていない新規フィールドのため
- **`wardRx` は `cnt` のみ編集可能。** `dose`・`ippoka` は詳細画面にも表示されていないため編集フォームにも出さず、保存時は読み込んだ時点の値をそのまま保持する
- フィールド単位のエラーハイライトは行わない。バリデーションエラーは画面上部に一覧表示する
- 同時編集の競合検出は行わない(本人のみが編集するため)

設計の詳細は [フェーズ2b-2 設計書](../specs/2026-08-26-phase2b2-edit-design.md) を参照。

- [ ] **Step 1: 保存関数を追加する**

`src/app/storage.js` の `getLog` の下に次を追加する。

```javascript
// Web画面からの編集を1日分だけ上書き保存する。移行(migration)と区別するため
// _meta.source を "web" にする。
async function saveLog(date, data) {
  var copy = {};
  Object.keys(data).forEach(function (key) { copy[key] = data[key]; });
  copy._meta = {
    source: "web",
    webEditedAt: new Date().toISOString()
  };
  await setDoc(doc(db, LOGS, date), copy);
}
```

  function esc(value) {

  function card(title, body, extraClass, titleClass) {

  function notesSection(log) {

      esc(log.notes) + "</textarea>");
  }

  // === 明細系(行の追加・削除ができる)セクション ===

  function fieldInputHtml(spec, value) {
    if (spec.type === "select") {
      var opts = spec.options.map(function (opt) {
        var label = (spec.optionLabels && spec.optionLabels[opt]) || opt;
        return '<option value="' + esc(opt) + '"' + (opt === value ? " selected" : "") +
          ">" + esc(label) + "</option>";
      }).join("");
      return '<select class="text-input" data-field="' + spec.field + '">' + opts + "</select>";
    }
    if (spec.type === "checkbox") {
      return '<input type="checkbox" data-field="' + spec.field + '"' + (value ? " checked" : "") + ">";
    }
    if (spec.type === "number") {
      return '<input type="number" min="0" step="1" class="num-input" data-field="' +
        spec.field + '" value="' + num(value) + '">';
    }
    return '<input type="text" class="text-input" data-field="' + spec.field + '" value="' + esc(value) + '">';
  }

  function listRowHtml(fields, item) {
    var cells = fields.map(function (spec) {
      return "<td>" + fieldInputHtml(spec, item ? item[spec.field] : undefined) + "</td>";
    }).join("");
    return "<tr data-row>" + cells +
      '<td><button type="button" class="btn btn-row-remove" data-remove-row>削除</button></td></tr>';
  }

  function listSectionHtml(sectionKey, list) {
    var def = LIST_SECTIONS[sectionKey];
    var head = def.fields.map(function (f) { return "<th>" + f.label + "</th>"; }).join("") + "<th></th>";
    var rows = (list || []).map(function (item) { return listRowHtml(def.fields, item); }).join("");
    var body = '<div class="tbl-wrap"><table class="tbl" data-section="' + sectionKey + '">' +
      "<thead><tr>" + head + "</tr></thead><tbody data-rows>" + rows + "</tbody></table></div>" +
      '<button type="button" class="btn" data-add-row="' + sectionKey + '">+ 行追加</button>';
    return card(def.title, body);
  }

  function collectListSection(container, sectionKey) {
    var def = LIST_SECTIONS[sectionKey];
    var tableEl = container.querySelector('table[data-section="' + sectionKey + '"]');
    var rows = tableEl.querySelectorAll("[data-row]");
    var items = [];
    rows.forEach(function (tr) {
      var item = {};
      def.fields.forEach(function (spec) {
        var input = tr.querySelector('[data-field="' + spec.field + '"]');
        if (spec.type === "checkbox") item[spec.field] = input.checked;
        else if (spec.type === "number") item[spec.field] = num(input.value);
        else item[spec.field] = input.value;
      });
      items.push(item);
    });
    return items;
  }

  // 「+ 行追加」「削除」は行数が動的に変わるため、イベント委譲で処理する。
  // body には render() のたびに作り直される #editBody を渡す(詳細は render() 側のコメント参照)。
  function attachListEvents(body) {
    body.addEventListener("click", function (event) {
      var addBtn = event.target.closest("[data-add-row]");
      if (addBtn) {
        var sectionKey = addBtn.getAttribute("data-add-row");
        var def = LIST_SECTIONS[sectionKey];
        var tbody = addBtn.closest(".card").querySelector("[data-rows]");
        tbody.insertAdjacentHTML("beforeend", listRowHtml(def.fields, null));
        return;
      }
      var removeBtn = event.target.closest("[data-remove-row]");
      if (removeBtn) {
        removeBtn.closest("tr").remove();
      }
    });
  }

  // === フォーム値の収集 ===
  // wardRx は cnt のみ編集可能。dose・ippoka は詳細画面にも表示されないため、

    (original.wardRx || []).forEach(function (w) { byWardRx[w.name] = w; });

    return {

      packerData: {

            '[data-packer-unit="san"][data-packer-item="分包紙"]').value),

      inquiries: collectListSection(container, "inquiries"),
      mrVisits: collectListSection(container, "mrVisits"),
      contacts: collectListSection(container, "contacts"),
      drugs: collectListSection(container, "drugs"),

  }

  // === 画面の組み立て ===

  function render(container, log, onSave, onCancel) {
    var normalized = root.DailyLog.normalize(log);

    container.innerHTML =
      '<div class="detail-head">' +
      '<button type="button" class="btn" id="editCancelBtn">キャンセル</button>' +
      '<span class="detail-title">' + esc(normalized.date) + "（編集中）</span>" +
      '<button type="button" class="btn btn-primary" id="editSaveBtn" style="margin-left:auto">保存</button>' +
      "</div>" +
      '<div id="editErrors"></div>' +
      '<div id="editBody">' +
      rxSection(normalized) +
      winjSection(normalized) +
      wardRxSection(normalized) +
      premixSection(normalized) +
      doctorChangesSection(normalized) +
      packerSection(normalized) +
      listSectionHtml("inquiries", normalized.inquiries) +
      listSectionHtml("mrVisits", normalized.mrVisits) +
      listSectionHtml("contacts", normalized.contacts) +
      listSectionHtml("drugs", normalized.drugs) +
      listSectionHtml("shortages", normalized.shortages) +
      listSectionHtml("meetings", normalized.meetings) +
      listSectionHtml("trainings", normalized.trainings) +
      notesSection(normalized) +
      "</div>";

    // #editBody は render() のたびに innerHTML ごと作り直されるため、ここに委譲用の
    // クリックリスナーを付ける。container(= 詳細画面の器)自体は編集を開くたびに
    // 使い回されるので、container に直接付けると開くたびにリスナーが積み重なってしまう。
    attachListEvents(container.querySelector("#editBody"));

import * as Auth from "./auth.js";
import * as Storage from "./storage.js";
import * as Migrate from "./migrate.js";

var el = function (id) { return document.getElementById(id); };
var parsed = null;
var currentMonth = null;
var currentLogs = [];
var loadSeq = 0;
var statusLoaded = false;

function esc(value) {
  return String(value === undefined || value === null ? "" : value)
    .replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;");
}

function describeError(err) {
  return err && err.message ? err.message : String(err);
}

function today() {
  var now = new Date();
  return { y: now.getFullYear(), m: now.getMonth() + 1 };
}

function todayDateStr() {
  var now = new Date();
  var pad2 = function (v) { return v < 10 ? "0" + v : "" + v; };
  return now.getFullYear() + "-" + pad2(now.getMonth() + 1) + "-" + pad2(now.getDate());
}

function render(state) {
  var signedOut = state.status === "signedOut";
  var denied = state.status === "denied";
  var allowed = state.status === "allowed";

  el("signedOut").hidden = !signedOut;
  el("deniedPanel").hidden = !denied;
  el("signedIn").hidden = !allowed;
  el("signInBtn").hidden = !signedOut;
  el("signOutBtn").hidden = signedOut;

  el("userLabel").textContent = allowed ? state.email : "";

  if (denied) {
    el("deniedDetail").textContent =
      state.email + " ではアクセスできません。許可されたアカウントでログインし直してください。";
  }
  if (allowed) {
    if (!currentMonth) currentMonth = today();
    showList();
  }
}

async function loadMonth() {
  var seq = ++loadSeq;
  var ym = currentMonth;

  el("monthLabel").textContent = window.Month.label(ym);
  el("listView").innerHTML = '<div class="empty">読み込み中…</div>';
  el("monthNotice").innerHTML = "";

  var logs;
  try {
    logs = await Storage.listLogsByMonth(ym);
  } catch (err) {
    if (seq !== loadSeq) return;
    el("listView").innerHTML = '<div class="card"><span class="err">読み取りに失敗しました: ' +
      describeError(err) + "</span></div>";
    return;
  }

  if (seq !== loadSeq) return;
  currentLogs = logs;

  window.ListView.render(el("listView"), ym, currentLogs, showDetail);

  var byDate = {};
  currentLogs.forEach(function (log) { byDate[log.date] = log; });
  var missing = window.Month.missingDays(
    window.Month.daysOf(ym.y, ym.m), byDate, todayDateStr());

  if (missing.length) {
    el("monthNotice").innerHTML = '<div class="notice notice-warn">記録の無い平日が' +
      missing.length + "日あります：" + missing.join("、") + "</div>";
  }
}

function showList() {
  el("listView").hidden = false;
  el("detailView").hidden = true;
  el("monthNotice").hidden = false;
  loadMonth();
}

function showDetail(date) {
  var log = currentLogs.filter(function (x) { return x.date === date; })[0];
  if (!log) return;
  el("listView").hidden = true;
  el("monthNotice").hidden = true;
  el("detailView").hidden = false;
  window.DetailView.render(el("detailView"), log, showList, function () { showEdit(date); });
  window.scrollTo(0, 0);
}

function showEdit(date) {
  var log = currentLogs.filter(function (x) { return x.date === date; })[0];
  if (!log) return;
  window.EditView.render(el("detailView"), log,
    function (data) { saveEdit(date, data); },
    function () { showDetail(date); });
  window.scrollTo(0, 0);
}

async function saveEdit(date, data) {
  var normalized = window.DailyLog.normalize(data);
  var errors = window.DailyLog.validate(normalized);
  var errorsEl = el("detailView").querySelector("#editErrors");

  if (errors.length) {
    errorsEl.innerHTML = '<div class="notice notice-err">入力内容にエラーがあります：<br>' +
      errors.map(function (e) { return esc(e.field) + "：" + esc(e.message); }).join("<br>") +
      "</div>";
    return;
  }

  var saveBtn = el("detailView").querySelector("#editSaveBtn");
  saveBtn.disabled = true;
  saveBtn.textContent = "保存中…";
  errorsEl.innerHTML = "";

  try {
    await Storage.saveLog(date, normalized);
  } catch (err) {
    errorsEl.innerHTML = '<div class="notice notice-err">保存に失敗しました: ' +
      describeError(err) + "</div>";
    saveBtn.disabled = false;
    saveBtn.textContent = "保存";
    return;
  }

  var existing = currentLogs.filter(function (x) { return x.date === date; })[0];
  if (existing) {
    currentLogs[currentLogs.indexOf(existing)] = normalized;
  } else {
    currentLogs.push(normalized);
  }
  showDetail(date);
}

async function refreshStatus() {
  el("dbStatus").textContent = "確認中…";
  try {
    var info = await Storage.countLogs();
    el("dbStatus").textContent = info.count === 0
      ? "データはまだありません。"
      : info.count + "件（" + info.first + " 〜 " + info.last + "）";
  } catch (err) {
    el("dbStatus").innerHTML = '<span class="err">読み取りに失敗しました: ' +
      describeError(err) + "</span>";
  }
}

el("prevMonthBtn").addEventListener("click", function () {

el("migrateFile").addEventListener("change", async function (event) {
  var file = event.target.files[0];
  parsed = null;
  el("migrateRunBtn").disabled = true;
  el("migrateResult").textContent = "";

  if (!file) {
    el("migratePreview").textContent = "";
    return;
  }

  el("migratePreview").textContent = "読み込み中…";
  try {
    var result = await Migrate.inspect(file);

    if (result.invalid.length) {
      el("migratePreview").innerHTML = '<span class="err">' +
        result.invalid.length + "件のレコードに問題があります。取り込みを中止しました。\n" +
        result.invalid.slice(0, 5).map(function (item) {
          return item.date + ": " + item.errors[0].field + " — " + item.errors[0].message;
        }).join("\n") + "</span>";
      return;
    }

    if (!result.logs.length) {
      el("migratePreview").innerHTML =
        '<span class="err">レコードが1件も読み取れませんでした。ファイルを確認してください。</span>';
      return;
    }

    parsed = result;
    var first = result.logs[0], last = result.logs[result.logs.length - 1];
    el("migratePreview").innerHTML = '<span class="ok">' +
      result.logs.length + "件を読み取りました（" + first.date + " 〜 " + last.date + "）。" +
      "医師" + result.masters.doctors.length + "名、予製" + result.masters.premixes.length + "品目。</span>";
    el("migrateRunBtn").disabled = false;
  } catch (err) {
    el("migratePreview").innerHTML = '<span class="err">読み込みに失敗しました: ' +
      describeError(err) + "</span>";
  }
});

el("migrateRunBtn").addEventListener("click", async function () {
  if (!parsed) return;
  el("migrateRunBtn").disabled = true;
  el("migrateResult").innerHTML = '<span class="progress">取り込み中… 0 / ' + parsed.logs.length + "</span>";

  try {
    var result = await Migrate.run(parsed, function (done, total) {
      el("migrateResult").innerHTML = '<span class="progress">取り込み中… ' + done + " / " + total + "</span>";
    });
    el("migrateResult").innerHTML = '<span class="ok">' + result.logs + "件を取り込みました。</span>";
    await refreshStatus();
    showList();
  } catch (err) {
    el("migrateResult").innerHTML = '<span class="err">取り込みに失敗しました: ' +
      describeError(err) + "</span>";
    el("migrateRunBtn").disabled = false;
  }
});

Auth.onChange(render);
